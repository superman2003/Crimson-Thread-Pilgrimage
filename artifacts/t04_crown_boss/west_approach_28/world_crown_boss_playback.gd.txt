func _finish() -> void:
    _release();_check(deaths==0,"crown west continuation survives this route attempt")
    print("CROWN_BOSS_STATE ",JSON.stringify({"pid":OS.get_process_id(),"wallet":scene.currency_count,"hp":scene.player.hp,"max_hp":scene.player.max_hp,"energy":scene.player.energy,"position":str(scene.player.position),"inventory":scene.inventory_counts,"checkpoint":scene.active_save_point_id,"deaths":deaths}))
    print("WORLD_C06_BOSS steps=%d failures=%d deaths=%d damage_events=%d heals=%d damage_suppression=false"%[steps,failures,deaths,damage_events,heals])
    preload("res://scripts/audio_retirement.gd").retire_tree(scene);scene.queue_free()
    for unused in 10:await process_frame
    preload("res://scripts/audio_director.gd")._stream_cache.clear();preload("res://scripts/audio_director.gd")._npc_voice_cache.clear()
    _check(await preload("res://scripts/audio_retirement.gd").drain(self),"crown Boss continuation retires audio")
    quit(0 if failures==0 else 1)

func _tick(count: int=1) -> void:
    for unused in count:
        if frames-crown_guard_at>=2:Input.action_release("bell_guard")
        if steps>=22 and steps<=26 and is_instance_valid(scene) and is_instance_valid(scene.player):
            var p: CharacterBody2D=scene.player
            for foe in get_nodes_in_group("enemy"):
                if not is_instance_valid(foe) or foe.spawn_id!="E09" or foe.hp<=0:continue
                if frames%8==0:print("CROWN_SALT_FOE ",JSON.stringify({"state":foe.state,"timer":foe.state_timer,"attack":foe.active_attack,"p":str(foe.position)}))
                if absf(foe.position.x-p.position.x)<260 and ((foe.state=="windup" and foe.state_timer<0.22) or foe.state=="active") and p.guard_cooldown<=0 and p.hurt_timer<=0 and p.attack_timer<=0 and p.energy>=8:
                    _axis(signf(foe.position.x-p.position.x));Input.action_press("bell_guard");crown_guard_at=frames
                    print("CROWN_SALT_ENEMY_GUARD ",p.position)
            for effect in get_nodes_in_group("hostile_attack"):
                if not is_instance_valid(effect) or effect.hit_player:continue
                if effect.position.distance_to(p.position-Vector2(0,42))<110 and p.guard_cooldown<=0 and p.hurt_timer<=0 and p.attack_timer<=0 and p.energy>=8:
                    _axis(signf(effect.position.x-p.position.x))
                    Input.action_press("bell_guard");crown_guard_at=frames
                    print("CROWN_SALT_NORMAL_GUARD ",p.position," effect=",effect.position)
        await super._tick()
        if is_instance_valid(scene) and is_instance_valid(scene.player):
            if observed_hp>0 and scene.player.hp<=0:observed_deaths+=1
            if (steps>=8 and steps<=12 or steps>=22 and steps<=25) and frames%4==0:
                var hits: Array=[]
                for i in scene.player.get_slide_collision_count():
                    var c: KinematicCollision2D=scene.player.get_slide_collision(i)
                    hits.append({"collider":str(c.get_collider().get_path()),"normal":str(c.get_normal()),"point":str(c.get_position())})
                print("CROWN_WEST_JUMP_TRACE ",JSON.stringify({"frame":frames,"position":str(scene.player.position),"velocity":str(scene.player.velocity),"air_available":scene.player.air_jump_available,"jump_input":Input.is_action_pressed("jump"),"hurt":scene.player.hurt_timer,"collisions":hits}))
            observed_hp=scene.player.hp
            deaths=observed_deaths

func _side_step_up(target: Vector2, outside: float) -> bool:
    steps+=1;_release();await process_frame;await process_frame;_axis(-1)
    Input.action_press("jump");await process_frame
    for frame in 100:
        _axis(-1)
        await _tick()
        if scene.player.position.x<outside:break
    _axis(0)
    for rise in 40:
        if scene.player.velocity.y>-70:break
        await _tick()
    Input.action_release("jump");await _tick(3)
    _check(scene.player.air_jump_available,"earned side ascent still has its real second jump")
    Input.action_press("jump");await process_frame
    var cleared:=false
    for frame in 200:
        if scene.player.position.y<target.y-5:cleared=true
        _axis(1 if cleared and scene.player.position.x<target.x-8 else 0)
        await _tick()
        if frame==24:Input.action_release("jump")
        if scene.player.is_on_floor() and absf(scene.player.position.x-target.x)<20 and absf(scene.player.position.y-target.y)<8:
            _release();print("CROWN_SIDE_ASCENT reached ",scene.player.position);return true
        if scene.player.position.x>target.x-10:_axis(0)
    _release();_check(false,"normal outside second jump blocked at "+str(scene.player.position));return false

func _narrow_obsidian_step() -> bool:
    steps+=1;_release();await process_frame;await process_frame
    _axis(1);Input.action_press("jump");await process_frame
    var second:=false
    var first_released:=false
    for frame in 240:
        var p: CharacterBody2D=scene.player
        _axis(1 if p.position.x<1590 else 0)
        if not first_released and p.velocity.y>=0:
            Input.action_release("jump");first_released=true;await process_frame
        if first_released and not second and p.position.x>1497 and p.position.x<1503 and p.air_jump_available and not p.is_on_wall():
            Input.action_press("jump");second=true;print("CROWN_OBSIDIAN_SECOND ",p.position);await process_frame
        if second and p.velocity.y>=0:Input.action_release("jump")
        await _tick()
        if p.is_on_floor() and absf(p.position.x-1600)<20 and absf(p.position.y-2030)<8:
            _release();print("CROWN_OBSIDIAN_STEP reached ",p.position);return true
        if frame%12==0:print("CROWN_OBSIDIAN_STEP ",JSON.stringify({"frame":frame,"p":str(p.position),"v":str(p.velocity),"second":second,"wall":p.is_on_wall()}))
    _release();_check(false,"normal obsidian outflow step blocked at "+str(scene.player.position));return false

func _salt_edge_punish() -> bool:
    steps+=1;_release();crown_flying_target="salt_edge_combat"
    await process_frame;await process_frame
    for frame in 360:
        var foe: Node2D=null
        for actor in get_nodes_in_group("enemy"):
            if is_instance_valid(actor) and actor.spawn_id=="E09" and actor.hp>0:foe=actor;break
        if foe==null:
            _release();crown_flying_target="";print("CROWN_SALT_ENEMY_DEFEATED on actual safe edge");return true
        var p: CharacterBody2D=scene.player
        var kind: String=String(foe.active_attack.get("type",""))
        var punish: bool=(foe.state=="windup" and kind!="guard" and foe.state_timer>0.25) or foe.state=="recover" or foe.state=="patrol"
        if frame%20==0:print("CROWN_SALT_EDGE_PUNISH ",JSON.stringify({"frame":frame,"player":str(p.position),"hp":p.hp,"energy":p.energy,"foe":str(foe.position),"foe_hp":foe.hp,"state":foe.state,"kind":kind}))
        if punish and absf(foe.position.x-p.position.x)<150 and p.attack_timer<=0 and p.hurt_timer<=0:
            if p.skill_cooldown<=0 and p.energy>=25:
                Input.action_press("skill_1");tower_skill_at=frames;await process_frame
            elif frames%20==0:
                Input.action_press("attack");attack_pressed_frame=frames
        await _tick()
        if p.hp<=0:break
    _release();crown_flying_target="";_check(false,"actual salt edge combat did not finish");return false
