extends "res://scripts/world_obsidian_tower_playback.gd"
## Normal earned Bride victory -> physical cable descent -> crown causeway.
const CROWN_ENTRY_SAVE := "user://test_world_c06_earned_entry.json"
var crown_elite_engaged:=false
const CROWN_ENTRY_SEED := "res://../artifacts/t04_bride_retry/earned_bride_seed.json"
func _run() -> void:
    for suffix in ["", ".bak", ".tmp"]:
        if FileAccess.file_exists(CROWN_ENTRY_SAVE+suffix):DirAccess.remove_absolute(ProjectSettings.globalize_path(CROWN_ENTRY_SAVE+suffix))
    var file:=FileAccess.open(CROWN_ENTRY_SAVE,FileAccess.WRITE)
    file.store_buffer(FileAccess.get_file_as_bytes(CROWN_ENTRY_SEED))
    file.close()
    scene=load("res://scenes/World.tscn").instantiate()
    scene.world_resume=true
    scene.world_save_path=CROWN_ENTRY_SAVE
    root.add_child(scene)
    await preload("res://scripts/world_ready_validation_helper.gd").wait_for(scene)
    current_scene=scene
    await _tick(15)
    _check(scene.currency_count==435 and scene.demo_complete and scene.inventory_counts.get("thread_dash",0)==0,"earned full Bride victory restores without unearned dash")
    var route: Array=[
        ["leap",4490,656,20,"physical mirror threshold after actual victory"],
        ["fight"],
        ["walk",4560,656],
        ["leap",4790,700,999,"ordinary jump through defeated Bride gate"],
        ["walk",5315,700],
        ["leap",5550,850,999,"ordinary jump onto first cable landing"],
        ["walk",5780,1140],
        ["walk",5580,1440],
        ["walk",5810,1745],
        ["walk",5600,2050],
        ["walk",5820,2355],
        ["walk",5610,2650],
        ["cable_fight"],
        ["walk",5730,2650],
        ["leap",6000,2724,20,"normal cable base to silent gate approach"],
        ["walk",6110,2724],
        ["leap",6330,2680,999,"ordinary jump to silent gate step"],
        ["leap",6650,2722,20,"actual exit dock"],
    ]
    for action in route:
        var ok:=true
        if action[0]=="cable_fight":
            ok=await _clear_cable_censor()
            if ok:await _settled_recover()
        elif action[0]=="leap":ok=await _leap(Vector2(action[1],action[2]),action[3],action[4])
        elif action[0]=="walk":ok=await _walk_without_jump(Vector2(action[1],action[2]))
        else:ok=await _clear_nearby()
        if not ok:await _finish();return
    await _crown_exit()
    if failures==0:await _crown_lower_loop()
    await _finish()
func _crown_exit() -> void:
    steps+=1
    await _world_interact()
    if scene.world_passage.active:
        _axis(signf(scene.world_passage.destination_point.x-scene.world_passage.source_point.x))
        for frame in 3600:
            await _tick()
            if not scene.world_passage.active:break
        _release()
    for frame in 2000:
        if not _transition_busy():break
        await process_frame
    await _tick(15)
    _check(String(scene.config.id)=="demo_ch06_silent_crown_core","actual crown causeway enters C6")
    print("CROWN_ENTRY_STATE ",JSON.stringify({"position":str(scene.player.position),"wallet":scene.currency_count,"inventory":scene.inventory_counts,"checkpoint":scene.active_save_point_id}))
func _finish() -> void:
    _release()
    Input.action_release("skill_1")
    _check(deaths==0,"actual crown entry survives without death")
    print("WORLD_C06_ENTRY steps=%d failures=%d deaths=%d damage_suppression=false"%[steps,failures,deaths])
    preload("res://scripts/audio_retirement.gd").retire_tree(scene)
    scene.queue_free()
    for unused in 10:await process_frame
    preload("res://scripts/audio_director.gd")._stream_cache.clear()
    preload("res://scripts/audio_director.gd")._npc_voice_cache.clear()
    _check(await preload("res://scripts/audio_retirement.gd").drain(self),"crown entry fixture retires audio")
    quit(0 if failures==0 else 1)

func _near_foe() -> Node2D:
    if crown_elite_engaged:return null
    if is_instance_valid(scene) and is_instance_valid(scene.player) and String(scene.config.id)=="demo_ch06_silent_crown_core" and scene.player.position.y>1600:
        for foe in get_nodes_in_group("enemy"):
            if is_instance_valid(foe) and foe.spawn_id=="E05" and foe.hp>0 and absf(foe.position.x-scene.player.position.x)<250 and absf(foe.position.y-scene.player.position.y)<240:return foe
    if is_instance_valid(scene) and is_instance_valid(scene.player) and String(scene.config.id)=="demo_ch05_obsidian_pilgrim_road" and scene.player.position.y>2300:
        for foe in get_nodes_in_group("enemy"):
            if is_instance_valid(foe) and foe.spawn_id=="E14" and foe.hp>0 and absf(foe.position.x-scene.player.position.x)<250 and absf(foe.position.y-scene.player.position.y)<240:return foe
    return super._near_foe()
func _clear_cable_censor() -> bool:
    steps+=1
    _release()
    var jumped_at: int=-100
    combat_enabled=true
    for frame in 600:
        var foe:=_near_foe()
        if foe==null:
            _release()
            combat_enabled=false
            return true
        var p: CharacterBody2D=scene.player
        var dx: float=foe.position.x-p.position.x
        _axis(signf(dx) if absf(dx)>90 or p.facing!=signi(dx) else 0)
        if frame-jumped_at>=25:Input.action_release("jump")
        if foe.position.y<p.position.y-90 and p.is_on_floor() and p.hurt_timer<=0 and p.attack_timer<=0 and frame-jumped_at>50:
            Input.action_press("jump");jumped_at=frame;await process_frame
        await _tick()
        if deaths>0:break
    combat_enabled=false
    _release()
    _check(false,"actual cable censor normal air attacks finish")
    return false

func _clear_rain_elite() -> bool:
    steps+=1
    crown_elite_engaged=true
    tower_spear=false
    combat_enabled=false
    _release()
    var jumped_at: int=-1000
    var doubled:=false
    var jump_kind:=""
    for frame in 1200:
        var foe: Node2D=null
        for actor in get_nodes_in_group("enemy"):
            if is_instance_valid(actor) and actor.spawn_id=="E01" and actor.hp>0:foe=actor;break
        if foe==null:
            crown_elite_engaged=false;_release();await _settled_recover();return true
        var p: CharacterBody2D=scene.player
        var state: String=String(foe.state)
        var kind: String=String(foe.active_attack.get("type",""))
        var side: float=signf(p.position.x-foe.position.x)
        if side==0:side=-1
        var punish: bool=state in ["recover","patrol"] or (state=="windup" and kind!="guard" and foe.state_timer>0.45)
        var goal: float=clampf(foe.position.x+side*(110 if punish else 195),4635,4855)
        _axis(signf(goal-p.position.x) if absf(goal-p.position.x)>7 else 0)
        if state=="windup" and foe.state_timer<0.25 and p.is_on_floor() and p.attack_timer<=0 and p.hurt_timer<=0 and (kind=="wave" or absf(foe.position.x-p.position.x)<170):
            Input.action_press("jump");jumped_at=frame;doubled=false;jump_kind=kind;await process_frame
        if frame-jumped_at==30:Input.action_release("jump");await process_frame
        if frame-jumped_at>=40 and frame-jumped_at<60 and jump_kind in ["wave","projectile"] and not doubled and p.air_jump_available and p.attack_timer<=0 and p.hurt_timer<=0:
            Input.action_press("jump");doubled=true;await process_frame
        if frame-jumped_at==75:Input.action_release("jump")
        var dx: float=foe.position.x-p.position.x
        if punish and p.is_on_floor() and absf(dx)<140 and p.attack_timer<=0 and p.hurt_timer<=0:
            _axis(signf(dx))
            if p.skill_cooldown<=0 and p.energy>=25:
                Input.action_press("skill_1");tower_skill_at=frames;await process_frame
            elif frames%12==0:
                Input.action_press("attack");attack_pressed_frame=frames
        for effect in get_nodes_in_group("hostile_attack"):
            if not is_instance_valid(effect) or effect.owner_actor!=foe or effect.hit_player:continue
            if effect.position.distance_to(p.position-Vector2(0,42))<95 and p.guard_cooldown<=0 and p.attack_timer<=0 and p.hurt_timer<=0 and p.energy>=8:
                _axis(signf(effect.position.x-p.position.x))
                Input.action_press("bell_guard");await process_frame;await _tick(2);Input.action_release("bell_guard")
        if frame%30==0:print("CROWN_ELITE ",JSON.stringify({"player":str(p.position),"hp":p.hp,"energy":p.energy,"foe_hp":foe.hp,"foe_position":str(foe.position),"state":state,"kind":kind,"remaining":foe.state_timer}))
        await _tick()
        if p.hp<=0:break
    crown_elite_engaged=false;_release();Input.action_release("skill_1")
    _check(false,"normal earned cleave punishes actual rain elite recovery")
    return false

func _settled_recover() -> void:
    _release()
    for frame in 90:
        if scene.player.attack_timer<=0 and scene.player.hurt_timer<=0 and scene.player.is_on_floor():break
        await _tick()
    if scene.player.hp>0 and scene.player.hp<scene.player.max_hp and scene.player.energy>=scene.player.heal_energy_cost:await _recover()

func _leap(target: Vector2,second_frame: int,label: String) -> bool:
    _release()
    for frame in 120:
        if scene.player.is_on_floor() and scene.player.hurt_timer<=0 and scene.player.attack_timer<=0:break
        await _tick()
    print("CROWN_JUMP_START ",JSON.stringify({"label":label,"position":str(scene.player.position),"floor":scene.player.is_on_floor(),"hurt":scene.player.hurt_timer,"air_jump":scene.player.air_jump_available}))
    return await super._leap(target,second_frame,label)
